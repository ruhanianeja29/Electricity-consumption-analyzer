// WattBot interaction functions
function toggleBot(){let w=$('botWindow');w.classList.toggle('show');$('botLabel').style.display=w.classList.contains('show')?'none':'block';if(w.classList.contains('show'))$('botInput').focus()}
function setBotLanguage(){let lang=$('botLang').value;let msg=lang==='English'?'Hello! 👋 I’m WattBot. Need help using the website? Choose a suggested question or type your own.':lang==='Hindi'?'नमस्ते! 👋 मैं WattBot हूँ। वेबसाइट इस्तेमाल करने में मदद चाहिए? नीचे सवाल चुनें या अपना सवाल लिखें।':'Heyyy! 👋 Main WattBot hoon. Website use karne mein help chahiye? Suggested questions try karo ya apna question type karo.';$('botWelcome').textContent=msg}
function addBotMessage(txt,who){let d=document.createElement('div');d.className='bot-msg'+(who==='user'?' user':'');d.textContent=txt;$('botMessages').appendChild(d);$('botMessages').scrollTop=$('botMessages').scrollHeight}
async function askSuggested(q){$('botInput').value=q;await sendBot()}
async function sendBot(){let q=$('botInput').value.trim();if(!q)return;addBotMessage(q,'user');$('botInput').value='';try{let d=await api('/api/bot',{question:q,lang:$('botLang').value});addBotMessage(d.reply,'bot')}catch(e){addBotMessage('Sorry, abhi response nahi aa paaya. Please try again.','bot')}}

// Shared API helper used by sendBot() (defined elsewhere in index.html)
async function api(path,body){let opt={method:body?'POST':'GET',cache:'no-store',headers:{}};const t=localStorage.getItem(TOKEN_KEY);if(t)opt.headers['X-Auth-Token']=t;if(body){opt.headers['Content-Type']='application/x-www-form-urlencoded';opt.body=new URLSearchParams(body)}let r=await fetch(path,opt);let raw=await r.text();let d;try{d=raw?JSON.parse(raw):{}}catch(e){throw new Error('This address is not serving the Home Energy C++ API. Open the exact URL printed in the terminal.')}if(r.status===401&&!/^\/api\/(login|register)/.test(path)){localStorage.removeItem(TOKEN_KEY);showAuthGate();throw new Error('Please log in again.')}if(!r.ok)throw new Error(d.error||('Server returned '+r.status));return d}
